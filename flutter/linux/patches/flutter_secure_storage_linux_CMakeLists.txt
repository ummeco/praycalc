cmake_minimum_required(VERSION 3.10)
set(PROJECT_NAME "flutter_secure_storage_linux")
project(${PROJECT_NAME} LANGUAGES CXX)

# This value is used when generating builds using this plugin, so it must
# not be changed
set(PLUGIN_NAME "flutter_secure_storage_linux_plugin")

add_definitions(-DAPPLICATION_ID="${APPLICATION_ID}")

add_library(${PLUGIN_NAME} SHARED
  "flutter_secure_storage_linux_plugin.cc"
)
apply_standard_settings(${PLUGIN_NAME})

# Use find_library instead of pkg_check_modules to avoid a cmake regression
# (FindPkgConfig.cmake ARGN bug) that causes 'Unknown arguments specified'
# when pkg_check_modules is called with version specifiers on cmake 3.28+.
find_library(LIBSECRET_LIBRARY secret-1)
find_path(LIBSECRET_INCLUDE_DIR libsecret/secret.h PATH_SUFFIXES libsecret-1)
if(NOT LIBSECRET_LIBRARY)
  message(FATAL_ERROR "libsecret-1 not found. Install: apt-get install libsecret-1-dev")
endif()

set_target_properties(${PLUGIN_NAME} PROPERTIES
  CXX_VISIBILITY_PRESET hidden)

target_compile_definitions(${PLUGIN_NAME} PRIVATE FLUTTER_PLUGIN_IMPL)
target_include_directories(${PLUGIN_NAME} INTERFACE
  "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(LIBSECRET_INCLUDE_DIR)
  include_directories(${LIBSECRET_INCLUDE_DIR})
endif()

target_link_libraries(${PLUGIN_NAME} PRIVATE flutter)
target_link_libraries(${PLUGIN_NAME} PRIVATE PkgConfig::GTK)
target_link_libraries(${PLUGIN_NAME} PRIVATE ${LIBSECRET_LIBRARY})

# List of absolute paths to libraries that should be bundled with the plugin
set(flutter_secure_storage_bundled_libraries
  ""
  PARENT_SCOPE
)
